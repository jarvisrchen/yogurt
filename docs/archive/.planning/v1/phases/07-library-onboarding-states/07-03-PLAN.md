```rust
async fn get_markdown(
    State(s): State<AppState>,
    Path(id): Path<String>,
) -> Result<(axum::http::HeaderMap, String), ApiError> {
    let m = s.meetings.get(&id).map_err(ApiError::Internal)?.ok_or(ApiError::NotFound)?;
    let path = s.exporter.path_for(&m);
    // Idempotent: write if missing (covers cases where the file was hand-deleted).
    if !path.exists() {
        s.exporter.write(&m).map_err(ApiError::Internal)?;
    }
    let body = std::fs::read_to_string(&path).map_err(|e| ApiError::Internal(e.into()))?;
    let mut h = axum::http::HeaderMap::new();
    h.insert(axum::http::header::CONTENT_TYPE, "text/markdown; charset=utf-8".parse().unwrap());
    Ok((h, body))
}

async fn reveal_in_finder(
    State(s): State<AppState>,
    Path(id): Path<String>,
) -> Result<StatusCode, ApiError> {
    let m = s.meetings.get(&id).map_err(ApiError::Internal)?.ok_or(ApiError::NotFound)?;
    let path = s.exporter.path_for(&m);
    if !path.exists() {
        s.exporter.write(&m).map_err(ApiError::Internal)?;
    }
    // macOS-specific. Only runs on darwin per project constraints.
    #[cfg(target_os = "macos")]
    {
        let status = std::process::Command::new("open")
            .arg("-R")
            .arg(&path)
            .status()
            .map_err(|e| ApiError::Internal(e.into()))?;
        if !status.success() {
            return Err(ApiError::Internal(anyhow::anyhow!("open -R failed: {status}")));
        }
    }
    Ok(StatusCode::NO_CONTENT)
}
```

```ts
export function useUpdateMeetingTitle() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: async ({ id, title }: { id: string; title: string }) => {
      const t = title.trim().length === 0 ? "Untitled meeting" : title.trim();
      return fetch(`/api/meetings/${id}`, {
        method: "PATCH",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({ title: t }),
      }).then(json<Meeting>);
    },
    onSuccess: (m) => {
      qc.invalidateQueries({ queryKey: meetingsKey });
      qc.setQueryData(meetingKey(m.id), m);
    },
  });
}

export async function copyMeetingMarkdown(id: string): Promise<void> {
  const md = await fetch(`/api/meetings/${id}/markdown`).then((r) => {
    if (!r.ok) throw new Error(`${r.status}`);
    return r.text();
  });
  await navigator.clipboard.writeText(md);
}

export async function revealMeetingInFinder(id: string): Promise<void> {
  const r = await fetch(`/api/meetings/${id}/reveal`, { method: "POST" });
  if (!r.ok) throw new Error(`${r.status}`);
}
```

```tsx
import { useEffect, useRef, useState } from "react";
import { useUpdateMeetingTitle } from "../../lib/api/meetings";

interface Props {
  id: string;
  title: string;
  className?: string;
}

export function InlineTitle({ id, title, className }: Props) {
  const [editing, setEditing] = useState(false);
  const [draft, setDraft] = useState(title);
  const ref = useRef<HTMLInputElement>(null);
  const update = useUpdateMeetingTitle();

  useEffect(() => { setDraft(title); }, [title]);
  useEffect(() => { if (editing) ref.current?.select(); }, [editing]);

  const commit = () => {
    const next = draft.trim().length === 0 ? "Untitled meeting" : draft.trim();
    if (next !== title) update.mutate({ id, title: next });
    setEditing(false);
  };

  if (!editing) {
    return (
      <span
        className={className}
        onDoubleClick={(e) => { e.preventDefault(); e.stopPropagation(); setEditing(true); }}
        title="Double-click to rename"
      >
        {title || "Untitled meeting"}
      </span>
    );
  }
  return (
    <input
      ref={ref}
      value={draft}
      onClick={(e) => e.stopPropagation()}
      onChange={(e) => setDraft(e.target.value)}
      onBlur={commit}
      onKeyDown={(e) => {
        if (e.key === "Enter") { e.preventDefault(); commit(); }
        else if (e.key === "Escape") { setDraft(title); setEditing(false); }
      }}
      className={`${className ?? ""} bg-white border border-[var(--blue)] rounded px-1 -mx-1 outline-none`}
    />
  );
}
```

```tsx
import { useState } from "react";
import { copyMeetingMarkdown, revealMeetingInFinder, useDeleteMeeting } from "../../lib/api/meetings";

interface Props { id: string }

export function MeetingCardActions({ id }: Props) {
  const [open, setOpen] = useState(false);
  const del = useDeleteMeeting();

  const onCopy = async (e: React.MouseEvent) => {
    e.preventDefault(); e.stopPropagation();
    setOpen(false);
    await copyMeetingMarkdown(id);
  };
  const onReveal = async (e: React.MouseEvent) => {
    e.preventDefault(); e.stopPropagation();
    setOpen(false);
    await revealMeetingInFinder(id);
  };
  const onDelete = async (e: React.MouseEvent) => {
    e.preventDefault(); e.stopPropagation();
    setOpen(false);
    if (window.confirm("Delete this meeting from the library? The markdown file in ~/.yogurt/notes/ stays put.")) {
      await del.mutateAsync(id);
    }
  };

  return (
    <div className="relative" onClick={(e) => e.stopPropagation()}>
      <button
        type="button"
        aria-label="Meeting actions"
        onClick={(e) => { e.preventDefault(); setOpen((o) => !o); }}
        className="px-2 py-1 text-[var(--mut)] hover:text-[var(--ink)]"
      >⋯</button>
      {open && (
        <div
          role="menu"
          className="absolute right-0 top-full mt-1 bg-white border border-[var(--line)] rounded-lg shadow-lg py-1 min-w-[180px] z-10 text-[13px]"
        >
          <button role="menuitem" onClick={onCopy}     className="block w-full text-left px-3 py-2 hover:bg-[var(--paper)]">Copy markdown</button>
          <button role="menuitem" onClick={onReveal}   className="block w-full text-left px-3 py-2 hover:bg-[var(--paper)]">Reveal in Finder</button>
          <button role="menuitem" onClick={onDelete}   className="block w-full text-left px-3 py-2 text-[var(--straw)] hover:bg-[var(--paper)]">Delete</button>
        </div>
      )}
    </div>
  );
}
```
