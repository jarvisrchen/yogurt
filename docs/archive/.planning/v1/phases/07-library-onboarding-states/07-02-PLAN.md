```sql
-- V004: FTS5 keyword search across notes + transcripts (LIB-07).

CREATE VIRTUAL TABLE meetings_fts USING fts5(
  title,
  notes_md,
  transcript_text,
  content='meetings',
  content_rowid='rowid',
  tokenize='unicode61 remove_diacritics 2'
);

-- Seed any existing rows (no-op on fresh DB).
INSERT INTO meetings_fts(rowid, title, notes_md, transcript_text)
SELECT rowid, title, notes_md, '' FROM meetings;

-- AFTER INSERT
CREATE TRIGGER meetings_ai AFTER INSERT ON meetings BEGIN
  INSERT INTO meetings_fts(rowid, title, notes_md, transcript_text)
  VALUES (new.rowid, new.title, new.notes_md, '');
END;

-- AFTER DELETE (use the contentless 'delete' command form)
CREATE TRIGGER meetings_ad AFTER DELETE ON meetings BEGIN
  INSERT INTO meetings_fts(meetings_fts, rowid, title, notes_md, transcript_text)
  VALUES ('delete', old.rowid, old.title, old.notes_md, '');
END;

-- AFTER UPDATE
CREATE TRIGGER meetings_au AFTER UPDATE ON meetings BEGIN
  INSERT INTO meetings_fts(meetings_fts, rowid, title, notes_md, transcript_text)
  VALUES ('delete', old.rowid, old.title, old.notes_md, '');
  INSERT INTO meetings_fts(rowid, title, notes_md, transcript_text)
  VALUES (new.rowid, new.title, new.notes_md, '');
END;
```

```ts
export const meetingsSearchKey = (q: string) => ["meetings", "search", q] as const;

export function useMeetingsSearch(q: string) {
  const trimmed = q.trim();
  return useQuery({
    queryKey: meetingsSearchKey(trimmed),
    queryFn: () =>
      fetch(`/api/meetings/search?q=${encodeURIComponent(trimmed)}`).then(json<Meeting[]>),
    staleTime: 5_000,
    enabled: trimmed.length > 0,
  });
}
```

2. Replace `web/src/components/library/SearchPill.tsx` (the plan-01 stub) with a real input:
```tsx
import { useEffect, useState } from "react";

interface SearchPillProps {
  value: string;
  onChange: (v: string) => void;
}

export function SearchPill({ value, onChange }: SearchPillProps) {
  // local state for snappy typing; debounced upward via parent
  const [local, setLocal] = useState(value);
  useEffect(() => { setLocal(value); }, [value]);
  useEffect(() => {
    const id = setTimeout(() => onChange(local), 200);
    return () => clearTimeout(id);
  }, [local, onChange]);

  return (
    <div role="search" className="flex items-center gap-2 bg-white border border-[var(--line)] rounded-full px-4 py-2 text-[13px] text-[var(--ink)] w-[280px]">
      <span aria-hidden className="text-[var(--mut)]">⌕</span>
      <input
        type="text"
        value={local}
        onChange={(e) => setLocal(e.target.value)}
        placeholder="Search notes & transcripts"
        className="flex-1 bg-transparent outline-none placeholder:text-[var(--mut)]"
      />
    </div>
  );
}
```
