2. Create `crates/yogurt-server/src/enhance.rs`. Adapted from Task 4.7 Step 2 with persistence + LLM-selection added:

```rust
use axum::{
    extract::{Path, State},
    http::StatusCode,
    Json,
};
use serde::Serialize;
use std::sync::Arc;

use crate::{
    llm_mock::MockLlm, llm_openai::OpenAiCompatClient, markdown_exporter::Meeting as ExpMeeting,
    AppState,
};
use yogurt_notes::merge_notes;
use yogurt_prompts::{EnhanceCtx, Mode as PromptMode, Prompts};

#[derive(Serialize)]
pub struct EnhanceResponse {
    pub enriched_md: String,
}

pub async fn enhance(
    State(state): State<Arc<AppState>>,
    Path(meeting_id): Path<String>,
) -> Result<Json<EnhanceResponse>, (StatusCode, String)> {
    // 1) Read meeting (notes_md + transcript_json + title + started_at)
    let snapshot = {
        let meetings = state.meetings.read().await;
        let m = meetings.get(&meeting_id)
            .ok_or((StatusCode::NOT_FOUND, format!("meeting {meeting_id} not found")))?;
        MeetingSnapshot {
            id: m.id.clone(),
            title: m.title.clone(),
            started_at_unix_ms: m.started_at_unix_ms,
            ended_at_unix_ms: m.ended_at_unix_ms,
            notes_md: m.notes_md.clone(),
            transcript_json: m.transcript_json.clone(),
        }
    };

    // 2) Load prompts
    let prompts = Prompts::load(prompt_mode(&state))
        .map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, format!("prompt load: {e}")))?;
    let user_prompt = prompts.render_enhance(&EnhanceCtx {
        notes: &snapshot.notes_md, transcript: &snapshot.transcript_json,
    }).map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, format!("render: {e}")))?;

    // 3) WS sending
    state.ws_broadcaster.send(&meeting_id, serde_json::json!({
        "type": "enhance_progress", "phase": "sending"
    })).await;

    // 4) LLM — try OpenAiCompat from env, fall back to MockLlm
    let llm_output = match OpenAiCompatClient::from_env() {
        Some(client) => client.complete("", &user_prompt).await
            .map_err(|e| (StatusCode::BAD_GATEWAY, format!("llm: {e}")))?,
        None => MockLlm.complete("", &user_prompt).await
            .map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, format!("mock: {e}")))?,
    };

    // 5) WS streaming + char count
    state.ws_broadcaster.send(&meeting_id, serde_json::json!({
        "type": "enhance_progress", "phase": "streaming", "chars": llm_output.len()
    })).await;

    // 6) Merge (server-side AST diff)
    let merged = merge_notes(&snapshot.notes_md, &llm_output, &snapshot.transcript_json)
        .map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, format!("merge: {e}")))?;
    let enriched_md = yogurt_notes::render::to_markdown(&merged);
    let enriched_doc_json = serde_json::to_string(&merged)
        .map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, format!("doc json: {e}")))?;

    // 7) Persist — in-memory + SQLite + per-meeting markdown
    {
        let mut meetings = state.meetings.write().await;
        if let Some(m) = meetings.get_mut(&meeting_id) {
            m.enriched_md = Some(enriched_md.clone());
            m.enriched_doc_json = Some(enriched_doc_json.clone());
        }
    }

    // SQLite — Phase 0 single-writer Mutex<Connection>
    {
        let writer = state.db_writer.lock().await;
        writer.execute(
            "UPDATE meetings SET enriched_md = ?, enriched_doc_json = ? WHERE id = ?",
            rusqlite::params![enriched_md, enriched_doc_json, &meeting_id],
        ).map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, format!("db: {e}")))?;
    }

    // Markdown file via MarkdownExporter (single writer to ~/.yogurt/notes/)
    state.markdown_exporter.write(&ExpMeeting {
        id: &snapshot.id,
        title: &snapshot.title,
        started_at_unix_ms: snapshot.started_at_unix_ms,
        ended_at_unix_ms: snapshot.ended_at_unix_ms,
        body_md: &enriched_md,
    }).map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, format!("md export: {e}")))?;

    // 8) WS done
    state.ws_broadcaster.send(&meeting_id, serde_json::json!({
        "type": "enhance_progress", "phase": "done"
    })).await;

    Ok(Json(EnhanceResponse { enriched_md }))
}

struct MeetingSnapshot {
    id: String, title: String,
    started_at_unix_ms: i64, ended_at_unix_ms: Option<i64>,
    notes_md: String, transcript_json: String,
}

fn prompt_mode(state: &AppState) -> PromptMode {
    match state.mode {
        crate::Mode::Dev => PromptMode::Dev,
        crate::Mode::Release => PromptMode::Release,
    }
}
```
