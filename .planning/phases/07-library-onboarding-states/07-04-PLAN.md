export function useSetFirstRunCompleted() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: (val: boolean) =>
      fetch("/api/settings", { method: "PATCH", headers: {"content-type":"application/json"}, body: JSON.stringify({ first_run_completed: val }) })
        .then((r) => { if (!r.ok) throw new Error(`${r.status}`); }),
    onSuccess: () => qc.invalidateQueries({ queryKey: ["settings"] }),
  });
}

3. Update `web/src/App.tsx` (replacing the plan-01 placeholder stub):
```tsx
import { BrowserRouter, Routes, Route, Navigate } from "react-router-dom";
import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import { Library } from "./routes/Library";
import { Welcome } from "./routes/Welcome";
import { Meeting } from "./routes/Meeting";
import { Settings } from "./routes/Settings";
import { useFirstRunRedirect } from "./hooks/useFirstRunRedirect";

const qc = new QueryClient();

function Shell() {
  useFirstRunRedirect();
  return (
    <Routes>
      <Route path="/" element={<Library />} />
      <Route path="/welcome" element={<Welcome />} />
      <Route path="/m/:id" element={<Meeting />} />
      <Route path="/settings" element={<Settings />} />
      <Route path="/starred" element={<Navigate to="/" replace />} />
      <Route path="*" element={<Navigate to="/" replace />} />
    </Routes>
  );
}

export function App() {
  return (
    <QueryClientProvider client={qc}>
      <BrowserRouter><Shell /></BrowserRouter>
    </QueryClientProvider>
  );
}
```
